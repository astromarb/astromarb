function contextFor(q){const a=answerSet(q)[0];const map={'American Government':'This question sits inside the structure, principles, rights, or responsibilities of U.S. government. Place the answer on a rough timeline and connect it to what came immediately before and after.','Symbols & Holidays':'This question tests shared national symbols and civic commemorations. Focus on what the symbol or holiday represents, rather than memorizing the label alone.'};return `<p>${map[q.category]}</p><p><b>Core recall:</b> ${esc(a||'Set dynamic answer in Settings')}. On the actual civics test, a concise correct answer is enough. For study, explain why it fits and connect it to one neighboring topic.</p><p><b>Memory hook:</b> “${esc(q.question.slice(0,70))}…” → “${esc(a||'dynamic answer')}.”</p>`}
