function renderQuizQuestion(){const q=quiz.qs[quiz.pos];document.getElementById('quiz').innerHTML=`<div class="card"><div class="row space"><span class="badge">Question ${quiz.pos+1} of 20</span><span class="timer" id="timer">${quiz.remaining}s</span></div><div class="progress" style="margin:14px 0"><div style="width:${quiz.pos/20*100}%"></div></div><div class="kicker">${esc(q.category)} • #${q.id}</div><div class="question">${esc(q.question)}</div><label>Your answer</label><input id="quizA" autocomplete="off"><button class="btn primary" id="submitQuiz" style="margin-top:12px">Submit answer</button><p class="muted small">A blank answer or expired timer is marked wrong.</p></div>`;document.getElementById('submitQuiz').onclick=()=>submitQuizAnswer(false);document.getElementById('quizA').onkeydown=e=>{if(e.key==='Enter')submitQuizAnswer(false)};document.getElementById('quizA').focus()}
async function submitQuizAnswer(expired){if(!quiz)return;clearInterval(timer);const q=quiz.qs[quiz.pos],input=document.getElementById('quizA'),text=expired?'':(input?input.value:''),r=expired?{ok:false,msg:'Time expired'}:scoreText(q,text),x=st(q.id);if(r.ok){quiz.correct++;x.correct++}else{quiz.wrong++;x.wrong++}quiz.results.push({id:q.id,question:q.question,answer:text,ok:!!r.ok,expired});persistStat(q.id);quiz.pos++;if(quiz.pos>=20)await finishQuiz();else{renderQuizQuestion();startTimer()}}
async function finishQuiz(){clearInterval(timer);const seconds=Math.round((Date.now()-quiz.started)/1000),payload={user_id:currentUser.id,session_id:currentSessionId,started_at:new Date(quiz.started).toISOString(),correct_count:quiz.correct,wrong_count:quiz.wrong,duration_seconds:seconds,passed:quiz.correct>=12,results:quiz.results};const {data,error}=await supabase.from('quiz_attempts').insert(payload).select().single();if(error){console.error(error);setSync('Sync error',true)}const attempt={id:data?.id||Date.now(),date:data?.completed_at||new Date().toISOString(),correct:quiz.correct,wrong:quiz.wrong,seconds,results:quiz.results};S.attempts.unshift(attempt);const pass=quiz.correct>=12;document.getElementById('quiz').innerHTML=`<div class="card"><div class="kicker">Attempt complete</div><h2 class="${pass?'goodtxt':'badtxt'}">${pass?'PASS':'NOT YET'} — ${quiz.correct} / 20</h2><p>Total time: ${Math.floor(seconds/60)}m ${seconds%60}s • Average ${Math.round(seconds/20)}s/question</p><div class="progress"><div style="width:${quiz.correct/20*100}%"></div></div><hr><button class="btn primary" id="again">Take another quiz</button></div><div class="card"><h3>Missed questions</h3>${quiz.results.filter(r=>!r.ok).map(r=>`<div class="item">#${r.id} ${esc(r.question)}</div>`).join('')||'<p class="goodtxt">Perfect score.</p>'}</div>`;quiz=null;document.getElementById('again').onclick=()=>{renderQuiz();document.getElementById('startQuiz').click()}}
