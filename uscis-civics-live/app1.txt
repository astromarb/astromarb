    supabase.from('study_sessions').select('*').eq('user_id',uid).order('started_at',{ascending:false}).limit(200),
    supabase.from('study_activity').select('question_id,studied_at,mode').eq('user_id',uid).order('studied_at',{ascending:false}).limit(5000),
    supabase.from('long_form_responses').select('*').eq('user_id',uid).order('created_at',{ascending:false}).limit(500)
  ]);
  const err=[settingsR,statsR,attemptsR,sessionsR,activityR,longR].find(r=>r.error)?.error;
  if(err){setSync('Sync error',true);console.error(err);return}
  S=base();
  if(settingsR.data){S.testDate=settingsR.data.test_date||'';S.minStudy=settingsR.data.daily_study_requirement||10;S.dynamicAnswers=settingsR.data.dynamic_answers||{};S.theme=settingsR.data.theme_preference||S.theme||'dark'}
  for(const x of statsR.data||[])S.stats[x.question_id]={seen:x.seen_count,correct:x.correct_count,wrong:x.wrong_count};
  S.attempts=(attemptsR.data||[]).map(a=>({id:a.id,date:a.completed_at,correct:a.correct_count,wrong:a.wrong_count,seconds:a.duration_seconds,results:a.results||[]}));
  S.sessions=(sessionsR.data||[]).map(x=>({id:x.id,kind:x.session_type,started:x.started_at,label:new Date(x.started_at).toLocaleString()}));

async function persistStat(id){
  if(!currentUser)return;const x=st(id);
  const {error}=await supabase.from('question_stats').upsert({user_id:currentUser.id,question_id:id,seen_count:x.seen,correct_count:x.correct,wrong_count:x.wrong,updated_at:new Date().toISOString()},{onConflict:'user_id,question_id'});
  if(error){console.error(error);setSync('Sync error',true)}
}

function scoreText(q,text){
  if(q.dynamic&&!S.dynamicAnswers[q.id])return {ok:false,score:0,msg:'Set your current answer for this dynamic question in Settings first.'};
  let n=norm(text);if(!n)return {ok:false,score:0,msg:'No answer entered.'};let matches=[];
  for(const a of answerSet(q)){let na=norm(a);if(na&&(n.includes(na)||na.includes(n))&&!matches.some(x=>norm(x)===na))matches.push(a)}
  const ok=matches.length>=q.required;return {ok,score:Math.min(1,matches.length/q.required),msg:ok?'Accepted':'Not enough required answer elements',matches};
}
